cmake_minimum_required(VERSION 3.26)
project(noise_iree C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Bytecode and weights are embedded by Rust; no file-backed IREE modules.
add_compile_definitions(IREE_FILE_IO_ENABLE=0)
# Optional test/CPU/compiler submodules are intentionally absent. The build script
# checks the pinned source and the two runtime dependencies we actually link.
set(IREE_ERROR_ON_MISSING_SUBMODULES OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_COMPILER OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(IREE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(IREE_ENABLE_LIBBACKTRACE OFF CACHE BOOL "" FORCE)
set(IREE_ENABLE_CPUINFO OFF CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_DEFAULTS OFF CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_VULKAN ON CACHE BOOL "" FORCE)
set(IREE_HAL_DRIVER_LOCAL_SYNC ON CACHE BOOL "" FORCE)
set(IREE_HAL_EXECUTABLE_LOADER_EMBEDDED_ELF ON CACHE BOOL "" FORCE)
set(IREE_HAL_EXECUTABLE_LOADER_DEFAULTS OFF CACHE BOOL "" FORCE)
set(IREE_HAL_EXECUTABLE_PLUGIN_DEFAULTS OFF CACHE BOOL "" FORCE)
if(MINGW)
  # Clang's MS extensions support IREE's Windows structured exception handler.
  add_compile_definitions(__USE_MINGW_ANSI_STDIO=1 WIN32_LEAN_AND_MEAN)
endif()
add_subdirectory("${IREE_SOURCE_DIR}" iree EXCLUDE_FROM_ALL)
if(MINGW)
  # GNU's linker does not honor MSVC pragma(comment, lib, ...).
  target_link_libraries(iree_async_async INTERFACE ws2_32)
  target_link_libraries(iree_base_threading_threading INTERFACE synchronization)
endif()
add_library(noise_iree SHARED noise_iree.c)
if(MINGW)
  target_sources(noise_iree PRIVATE sysv_win64.c)
endif()
target_link_libraries(noise_iree PRIVATE iree::runtime::runtime iree::hal::drivers::vulkan iree::hal::drivers::local_sync::sync_driver iree::hal::local::loaders::embedded_elf_loader)
set_target_properties(noise_iree PROPERTIES C_VISIBILITY_PRESET hidden)
if(MINGW)
  target_link_options(noise_iree PRIVATE -static-libgcc -static-libstdc++)
endif()
